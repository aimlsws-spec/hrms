                        <script>
                            (function () {
                                fetch('/api/leave/holidays')
                                    .then(r => r.json())
                                    .then(data => {
                                        const container = document.getElementById('upcomingHolidaysContainer');
                                        if (data.success && data.holidays) {
                                            const now = new Date();
                                            const upcoming = data.holidays.filter(h => new Date(h.holiday_date) > now).slice(0, 3);
                                            if (upcoming.length === 0) {
                                                container.innerHTML = '<p style="color: #64748b; text-align: center;">No upcoming holidays</p>';
                                            } else {
                                                const colors = [{ color: '#fef3c7', border: '#f59e0b', text: '#92400e' }, { color: '#dbeafe', border: '#3b82f6', text: '#1e3a8a' }, { color: '#fce7f3', border: '#ec4899', text: '#831843' }];
                                                container.innerHTML = upcoming.map((h, i) => {
                                                    const c = colors[i % 3];
                                                    const d = new Date(h.holiday_date);
                                                    const dateStr = d.toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' });
                                                    return `<div style="padding: 12px; background: ${c.color}; border-radius: 10px; margin-bottom: 8px; border-left: 3px solid ${c.border};"><div style="font-weight: 600; color: ${c.text}; font-size: 14px;">${h.holiday_name}</div><div style="font-size: 12px; color: ${c.text}; margin-top: 2px;">&#128197; ${dateStr}</div></div>`;
                                                }).join('');
                                            }
                                        } else {
                                            container.innerHTML = '<p style="color: #64748b; text-align: center;">No upcoming holidays</p>';
                                        }
                                    })
                                    .catch(() => {
                                        document.getElementById('upcomingHolidaysContainer').innerHTML = '<p style="color: #64748b; text-align: center;">No upcoming holidays</p>';
                                    });
                            })();
